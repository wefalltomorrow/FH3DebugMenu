cmake_minimum_required(VERSION 3.20)
project(FH3NativeCameraDiscovery LANGUAGES CXX)
add_library(FH3NativeCameraDiscovery STATIC CameraObjectLocator.cpp)
target_compile_features(FH3NativeCameraDiscovery PUBLIC cxx_std_20)
target_include_directories(FH3NativeCameraDiscovery PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
add_executable(FH3NativeCameraDiscoveryTests test_CameraObjectLocator.cpp)
target_link_libraries(FH3NativeCameraDiscoveryTests PRIVATE FH3NativeCameraDiscovery)
enable_testing()
add_test(NAME CameraObjectLocator COMMAND FH3NativeCameraDiscoveryTests)
if(MSVC)
  target_compile_options(FH3NativeCameraDiscovery PRIVATE /W4 /permissive-)
  target_compile_options(FH3NativeCameraDiscoveryTests PRIVATE /W4 /permissive-)
else()
  target_compile_options(FH3NativeCameraDiscovery PRIVATE -Wall -Wextra -Wpedantic)
  target_compile_options(FH3NativeCameraDiscoveryTests PRIVATE -Wall -Wextra -Wpedantic)
endif()
