cmake_minimum_required(VERSION 3.20)
project(FH3NativeCameraCore VERSION 0.1 LANGUAGES CXX)
add_library(FH3NativeCameraCore STATIC CameraModeBridge.cpp)
target_compile_features(FH3NativeCameraCore PUBLIC cxx_std_20)
target_include_directories(FH3NativeCameraCore PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
add_executable(FH3NativeCameraCoreTests test_CameraModeBridge.cpp)
target_link_libraries(FH3NativeCameraCoreTests PRIVATE FH3NativeCameraCore)
enable_testing()
add_test(NAME CameraModeCore COMMAND FH3NativeCameraCoreTests)
if(MSVC)
    target_compile_options(FH3NativeCameraCore PRIVATE /W4 /permissive-)
    target_compile_options(FH3NativeCameraCoreTests PRIVATE /W4 /permissive-)
else()
    target_compile_options(FH3NativeCameraCore PRIVATE -Wall -Wextra -Wpedantic)
    target_compile_options(FH3NativeCameraCoreTests PRIVATE -Wall -Wextra -Wpedantic)
endif()
